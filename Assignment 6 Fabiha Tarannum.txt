     if( a == null ) {
     return 0 ; }
     
     int data = (Int)a.elem ;
      int b = 0 ;
     
     if( level == 0) {
     b = data  ;}
     
     else {
     b = data % level ; }
      
     Int lft = helper ( a.lft , level+1) ;
      Int rht = helper ( a.rht , level+1) ;
      return b + lft + rht ; }}

public class TaskFour {
  public static void swapChild( BTNode head , int b , int level  ) {
   if( head == null || level >= b ) {
      return;}
     BTNode a = head.lft ;
     
     head.lft = head.rht ;
     
     head.rht = a;
     
     swapChild( head.lft , level + 1 , b ) ;
     
     swapChild( head.rht , level+1 , b ) ;  } }
